# 01 数据模型（完整建表 SQL）

> 全部表由 relay-store 启动时执行（`CREATE TABLE IF NOT EXISTS`）。本文档是**唯一权威 schema**，改表必须同步更新本文档与 store 的 SCHEMA 常量。

## 1. 用户与分组

```sql
-- 用户（角色：admin 管理员 / operator 运营 / user 普通乘客）
CREATE TABLE IF NOT EXISTS users (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  username TEXT UNIQUE NOT NULL,
  password_hash TEXT NOT NULL,          -- scrypt
  display_name TEXT DEFAULT '',
  email TEXT DEFAULT '',
  phone TEXT DEFAULT '',
  role TEXT NOT NULL DEFAULT 'user',
  group_id INTEGER NOT NULL DEFAULT 1,  -- 所属分组（渠道按组授权）
  quota_limit INTEGER NOT NULL DEFAULT -1,   -- 总额度（token 数），-1 不限
  used_quota INTEGER NOT NULL DEFAULT 0,
  status INTEGER NOT NULL DEFAULT 1,    -- 1 启用 / 0 禁用
  created_at TEXT NOT NULL
);

-- 分组（NEWAPI 风格：渠道按组授权 + 计费倍率）
CREATE TABLE IF NOT EXISTS groups (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  name TEXT UNIQUE NOT NULL,
  quota_multiplier REAL NOT NULL DEFAULT 1.0,  -- 组内用户计费倍率（1.0 = 原价）
  remark TEXT DEFAULT '',
  created_at TEXT NOT NULL
);
-- 默认分组（id=1，必须存在）
INSERT OR IGNORE INTO groups (id, name, quota_multiplier) VALUES (1, 'default', 1.0);
```

## 2. 渠道

```sql
CREATE TABLE IF NOT EXISTS channels (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  name TEXT NOT NULL,                    -- 渠道显示名
  type TEXT NOT NULL,                    -- openai/deepseek/qwen/zhipu/kimi/minimax/volcengine/
                                         -- qianfan/hunyuan/groq/together/mistral/xai/openrouter/
                                         -- siliconflow/ollama/azure/anthropic/gemini/spark/cohere
  base_url TEXT NOT NULL,
  api_key_enc TEXT NOT NULL,             -- 加密存储（见下），绝不明文
  models TEXT DEFAULT '',                -- 支持的模型（逗号分隔；空 = 全部）
  model_mapping TEXT DEFAULT '{}',       -- JSON：{"gpt-4o":"deepseek-chat"}
  priority INTEGER NOT NULL DEFAULT 0,   -- 优先级：越大越先选（NEWAPI 同款）
  weight INTEGER NOT NULL DEFAULT 1,     -- 同优先级内轮询权重
  group_ids TEXT NOT NULL DEFAULT '1',   -- 授权分组（逗号分隔：1,2）
  status INTEGER NOT NULL DEFAULT 1,     -- 1 启用 / 0 禁用
  auto_ban INTEGER NOT NULL DEFAULT 1,   -- 连续失败自动禁用（1 开）
  max_retries INTEGER NOT NULL DEFAULT 1,-- 失败重试次数（每个请求）
  timeout_ms INTEGER NOT NULL DEFAULT 60000,
  created_at TEXT NOT NULL
);

-- 索引：按优先级+权重选渠道
CREATE INDEX IF NOT EXISTS idx_channels_status ON channels(status, priority DESC, weight DESC);
```

**API Key 加密**：`api_key_enc = AES-GCM(base64(随机 nonce) + base64(ciphertext))`，主密钥来自 `RELAY_MASTER_KEY` 环境变量（首次启动生成并提示保存，或直接指定 32 字节）。relay-core 提供 `crypto::encrypt/decrypt`。

## 3. 令牌（客户端凭证）

```sql
CREATE TABLE IF NOT EXISTS tokens (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  name TEXT NOT NULL,
  key_hash TEXT UNIQUE NOT NULL,         -- sha256(明文 sk-xxx)，明文只展示一次
  status INTEGER NOT NULL DEFAULT 1,     -- 1 启用 / 0 禁用
  quota_limit INTEGER NOT NULL DEFAULT -1,  -- 令牌级额度
  used_quota INTEGER NOT NULL DEFAULT 0,
  expired_at TEXT,                       -- NULL = 永不过期
  created_at TEXT NOT NULL
);
CREATE INDEX IF NOT EXISTS idx_tokens_hash ON tokens(key_hash);
```

## 4. 兑换码（NEWAPI 同款）

```sql
CREATE TABLE IF NOT EXISTS redeem_codes (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  code TEXT UNIQUE NOT NULL,             -- REDEEM-XXXX-XXXX-XXXX-XXXX
  quota INTEGER NOT NULL,                -- 可兑换额度（token 数）
  status INTEGER NOT NULL DEFAULT 0,     -- 0 未使用 / 1 已使用 / 2 禁用
  used_by INTEGER,                       -- 使用者 user_id
  used_at TEXT,
  expires_at TEXT,
  remark TEXT DEFAULT '',
  created_at TEXT NOT NULL
);
CREATE INDEX IF NOT EXISTS idx_redeem_status ON redeem_codes(status);
```

## 6. 用量日志

```sql
CREATE TABLE IF NOT EXISTS usage_logs (
  id INTEGER PRIMARY KEY AUTOINCREMENT,

  group_id INTEGER NOT NULL DEFAULT 1,
  channel_id INTEGER NOT NULL,
  model TEXT NOT NULL,
  prompt_tokens INTEGER NOT NULL DEFAULT 0,
  completion_tokens INTEGER NOT NULL DEFAULT 0,

CREATE INDEX IF NOT EXISTS idx_usage_created ON usage_logs(created_at);
```

## 8. 通知 / 支付 / 充值

```sql
-- 外部接入配置（密钥加密）
CREATE TABLE IF NOT EXISTS notify_config (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  kind TEXT UNIQUE NOT NULL,             -- sms_aliyun / sms_tencent / email_smtp /
                                         -- pay_alipay / pay_wechat / qr_manual
  config_enc TEXT NOT NULL,              -- JSON 加密（含 AccessKey/Secret/SMTP 账号等）
  enabled INTEGER NOT NULL DEFAULT 1,
  updated_at TEXT NOT NULL
);

-- 充值订单
CREATE TABLE IF NOT EXISTS recharge_orders (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  order_no TEXT UNIQUE NOT NULL,         -- R20260826xxxxxxxx（时间戳+随机）
  user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  amount REAL NOT NULL,                  -- 支付金额（元）
  quota INTEGER NOT NULL,                -- 到账额度（token 数，按充值档位）
  pay_method TEXT NOT NULL,              -- alipay / wechat / manual
  status TEXT NOT NULL DEFAULT 'pending',-- pending / paid / confirmed / cancelled
  qr_url TEXT,                           -- 收款码图片 URL / 支付链接
  paid_at TEXT,
  confirmed_by INTEGER,                  -- 手动确认的管理员 user_id
  created_at TEXT NOT NULL
);
CREATE INDEX IF NOT EXISTS idx_recharge_user ON recharge_orders(user_id, created_at);

-- 通知发送记录
CREATE TABLE IF NOT EXISTS notification_logs (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id INTEGER,
  channel TEXT NOT NULL,                 -- sms / email
  template TEXT NOT NULL,                -- quota_low / quota_exhausted / payment_success / channel_down
  to_addr TEXT,                          -- 手机号 / 邮箱
  status TEXT NOT NULL,                  -- success / failed
  error TEXT,
  created_at TEXT NOT NULL
);
```

## 10. 级联删除策略（与 DocPI 的 bug 对照）

- `users` 删除 → 级联删 `tokens`、`recharge_orders`（外键 `ON DELETE CASCADE`）
- `channels` 删除 → **不级联**删 `usage_logs`（日志保留，channel_id 存 id 即可）
- **本项目所有删除都遵循：日志永远不删**（审计要求）
- SQLite 需执行 `PRAGMA foreign_keys = ON;`（rusqlite 每次连接设置）

## 11. Store 层结构

```rust
// crates/relay-store/src/lib.rs
pub trait Store: Send + Sync {
    // 用户/分组
    fn get_user_by_id(&self, id: i64) -> Result<Option<User>>;
    fn get_user_by_username(&self, name: &str) -> Result<Option<User>>;
    fn create_user(&self, u: &User) -> Result<i64>;
    fn update_user(&self, u: &User) -> Result<()>;
    fn delete_user(&self, id: i64) -> Result<()>;
    fn list_users(&self, page: u32, size: u32) -> Result<(Vec<User>, u32)>;
    fn get_group(&self, id: i64) -> Result<Option<Group>>;
    // 渠道
    fn list_channels(&self, enabled_only: bool) -> Result<Vec<Channel>>;
    fn get_channel(&self, id: i64) -> Result<Option<Channel>>;
    fn upsert_channel(&self, c: &Channel) -> Result<i64>;
    fn delete_channel(&self, id: i64) -> Result<()>;
    fn update_channel_status(&self, id: i64, status: i32) -> Result<()>;
    // 令牌
    fn get_token_by_hash(&self, hash: &str) -> Result<Option<Token>>;
    fn create_token(&self, t: &Token) -> Result<i64>;
    fn delete_token(&self, id: i64) -> Result<()>;
    fn add_used_quota(&self, token_id: i64, user_id: i64, delta: i64) -> Result<()>;
    // 兑换码
    fn get_redeem_code(&self, code: &str) -> Result<Option<RedeemCode>>;
    fn redeem(&self, code_id: i64, user_id: i64) -> Result<()>;  // 事务：码状态 + 用户额度
    // 用量
    fn insert_usage(&self, u: &UsageLog) -> Result<()>;
    fn query_usage(&self, filter: &UsageFilter) -> Result<Vec<UsageLog>>;
    fn usage_stats(&self, filter: &UsageFilter) -> Result<StatSummary>;
    // 其他
    fn upsert_pricing(&self, p: &Pricing) -> Result<()>;
    fn get_pricing(&self, model: &str) -> Result<Option<Pricing>>;
    fn save_tokenizer_file(&self, f: &TokenizerFile) -> Result<()>;
    fn get_tokenizer_file(&self, family: &str) -> Result<Option<TokenizerFile>>;
    fn get_setting(&self, key: &str) -> Result<Option<String>>;
    fn set_setting(&self, key: &str, value: &str) -> Result<()>;
}

// 实现：SqliteStore { conn: Mutex<Connection> } —— 单连接 + 全局锁（SQLite 写串行，够用）
// 事务方法（redeem / add_used_quota）用 conn.transaction() 包住
```
